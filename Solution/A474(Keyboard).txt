#include <iostream>
#include <string>

using namespace std;

int main()
{
	/*
	qwertyuiop
	asdfghjkl;
	zxcvbnm,./
	*/
	char p;
	string s, sample = "qwertyuiopasdfghjkl;zxcvbnm,./";
	int x = 1;
	cin >> p >> s;
	if (p == 'R')
		x = -1;
	for (int i = 0; i < s.length(); i++)
	{
		for (int j = 0; j < sample.length(); j++)
		{
			if (s[i] == sample[j])
			{
				cout << sample[j + x];
				break;
			}
		}
	}
	return 0;
}
