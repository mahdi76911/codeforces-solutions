#include <iostream>

using namespace std;

int main()
{
	int n, a[1000001], x, m, q, c = 1;
	cin >> n;
	for (int i = 1; i <= n; i++)
	{
		cin >> x;
		for (int j = 0; j < x; j++)
			a[j + c] = i;
		c += x;
	}
	cin >> m;
	for (int i = 0; i < m; i++)
	{
		cin >> q;
		cout << a[q] << endl;
	}
	return 0;
}
