#include <iostream>
#include <string>

using namespace std;

int main()
{
	string s;
	bool b = true;
	cin >> s;
	for (int i = 1; i < s.length(); i++)
		if (s[i] >= 'a' && s[i] <= 'z')
		{
			b = false;
			break;
		}
	if (b)
	{
		for (int i = 0; i < s.length(); i++)
			if (s[i] >= 'A' && s[i] <= 'Z')
				s[i] -= 'Z' - 'z';
			else
			{
				s[i] += 'Z' - 'z';
			}
	}
	cout << s;
	return 0;
}
