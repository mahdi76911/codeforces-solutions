#include<iostream>
#include<string>
#include <algorithm>

using namespace std;

int main()
{
int c=0;
string s;
cin>>s;
for (int i=0;i<s.length();i++)
{
if(isupper(s[i]))
c++;
else
c--;
}
if(c>0)
transform(s.begin(), s.end(), s.begin(), ::toupper);
else
transform(s.begin(), s.end(), s.begin(), ::tolower);
cout<<s;
return 0;
}
