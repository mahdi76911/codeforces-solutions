#include <iostream>

using namespace std; 

int main() 
{
	int m, s, ms, Ms;
	char mn[101] = { '0' }, Mn[101] = { '0' };
	cin >> m >> s;
	ms = s;
	Ms = s;
	if (s > 9 * m)
	{
		cout << "-1 -1";
		return 0;
	}
	if (s == 0)
		if (m == 1)
			cout << "0 0";
		else
			cout << "-1 -1";
	else
	{
		for (int i = 0; i < m; i++)
		{
			if (Ms > 9)
			{
				Mn[i] = '9';
				Ms -= 9;
			}
			else
			{
				Mn[i] = '0' + Ms;
				Ms = 0;
			}
		}
		if (ms <= 9 * (m - 1))
		{
			mn[0] = '1';
			ms--;
			for (int i = m-1; i >= 1; i--)
			{
				if (ms > 9)
				{
					mn[i] = '9';
					ms -= 9;
				}
				else
				{
					mn[i] = '0' + ms;
					ms = 0;
				}
			}
		}
		else
		{
			mn[0] = '0' + ms - 9 * (m - 1);
			for (int i = m-1; i >= 1; i--)
			{
				mn[i] = '9';
			}
		}
		cout << mn << " " << Mn;
	}
	return 0;
}
