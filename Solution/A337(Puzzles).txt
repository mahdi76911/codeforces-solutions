#include <iostream>

using namespace std;

int main()
{
	int n, m, res;
	int f[50];
	cin >> n >> m;
	for (int i = 0; i < m; i++)
		cin >> f[i];
	for (int i = 1; i < m; i++)
	{
		int c = i;
		while (c > 0)
		{
			if (f[c] < f[c - 1])
			{
				swap(f[c], f[c - 1]);
				c--;
			}
			else
				break;
		}
	}
	res = f[m - 1] - f[0];
	for (int i = 0; i <= m - n; i++)
	{
		if (res > f[i + n - 1] - f[i])
			res = f[i + n - 1] - f[i];
	}
	cout << res;
	return 0;
}
