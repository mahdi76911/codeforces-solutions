#include <iostream>
#include <algorithm>

using namespace std;

int main()
{
	int n, a, c = 0, x = 0, m = 0, r = -1;

	cin >> n;
	while (n--)
	{
		cin >> a;
		c += a;
		
		x += 1 - a * 2;
		r = max(r, x - m);
		m = min(x, m);
	}	
	cout << r + c;
	return 0;
}

#include <iostream>

using namespace std;

int SubSeriesSumMax(int a[], int s, int e)	//max ... beyne azaye s ta khod e serie a 
{
	if (s == e)
		return a[s];
	if (a[s] < 0)
		return SubSeriesSumMax(a, s + 1, e);
	//bakhshe cut kardan navar
	for (int i = s + 1; i < e; i += 2)
	{
		if (a[i]<0 && fabs(a[i]) > a[i - 1] && fabs(a[i]) > a[i + 1])
			return fmax(SubSeriesSumMax(a, s, i - 1), SubSeriesSumMax(a, i + 1, e));
	}
	//bakhshe jam kardane greedy




	return 0;
}
