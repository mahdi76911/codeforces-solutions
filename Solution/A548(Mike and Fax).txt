#include <iostream>
#include <string>

using namespace std;

bool palindrome(string s)
{
	for (int i = 0; i < s.length() / 2; i++)
	{
		if (s[i] != s[s.length() - (i + 1)])
			return false;
	}
	return true;
}

int main()
{
	int n;
	string s;
	cin >> s >> n;
	if (s.length() % n != 0)
	{
		cout << "NO";
		return 0;
	}
	for (int i = 0; i < n; i++)
	{
		if (!palindrome(s.substr(i*(s.length() / n), s.length() / n)))
		{
			cout << "NO";
			return 0;
		}
	}
	cout << "YES";
	return 0;
}
