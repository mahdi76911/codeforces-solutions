#include <iostream>

using namespace std;

int main()
{
	int n, x;
	int a[] = { 0,0,0 };
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> x;
		a[i % 3] += x;
	}
	if (a[0] > a[1] && a[0] > a[2])
		cout << "chest";
	else if (a[1] > a[0] && a[1] > a[2])
		cout<<"biceps";
	else if (a[2] > a[1] && a[2] > a[0])
		cout<<"back";
	return 0;
}
