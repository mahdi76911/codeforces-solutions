#include <iostream>

using namespace std;

int main()
{
	int a = 0, b = 0, n;
	for (int i = 0; i < 3; i++)
	{
		cin >> n;
		a += n;
	}
	for (int i = 0; i < 3; i++)
	{
		cin >> n;
		b += n;
	}
	cin >> n;
	if (ceil((double)a / 5) + ceil((double)b / 10) <= n)
		cout << "YES";
	else
		cout << "NO";
	return 0;
}
