#include <iostream>
#include <string>

using namespace std;

int main()
{
	int res = 0;
	bool l[26];
	for (int i = 0; i < 26; i++)
		l[i] = true;
	string s;
	getline(cin, s);
	for (int i = 1; i < s.length() - 1; i += 3)
	{
		if (l[s[i] - 'a'])
		{
			res++;
			l[s[i] - 'a'] = false;
		}
	}
	cout << res;
	return 0;
}
