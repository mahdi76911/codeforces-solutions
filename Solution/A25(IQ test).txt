#include <iostream>

using namespace std;

int main()
{
	int n, a[100], x = 0;
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> a[i];
		a[i] = a[i] % 2;
		x += a[i];
	}
	for (int i = 0; i < n; i++)
	{
		if ((x == 1 && a[i] % 2 == 1) || (x != 1 && a[i] % 2 == 0))
		{
			cout << i + 1;
			break;
		}
	}
	return 0;
}
