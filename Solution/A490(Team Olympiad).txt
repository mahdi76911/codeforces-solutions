#include <iostream>
#include <algorithm>

using namespace std;

int main()
{
int n, a[3]={0,0,0}, x;
int s[3][5000];
cin>>n;
for(int i=0;i<n;i++)
{
cin>>x;
s[x-1][a[x-1]] = i+1;
a[x-1]++;
}
cout<<min(min(a[0],a[1]),a[2])<<endl;
for(int i=0;i<min(min(a[0],a[1]),a[2])!=0;i++)
{
cout<<s[0][i]<<" "<<s[1][i]<<" "<<s[2][i]<<endl;
}
return 0;
}
