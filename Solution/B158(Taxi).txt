#include <iostream>

using namespace std;

int main()
{
    int n,x;
    int c[5]={0,0,0,0,0};
    cin>>n;
    for (int i=0;i<n;i++)
    {
        cin>>x;
        c[x]++;
    }
    c[0]=c[4]+c[3]+(c[2]/2);
    c[1]=c[1]-c[3];
    if(c[1]>0)
    {
        c[0]+=(c[1]/4);
        c[1]=c[1]%4;
    }
    if(c[2]%2!=0)
    {
        c[1]=c[1]-2;
        c[0]++;
    }
    if(c[1]>0)
        c[0]++;
    cout<<c[0];
    return 0;
}
