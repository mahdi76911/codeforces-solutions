#include <iostream>
#include <math.h>

using namespace std;

int main()
{
	//Sheldon, Leonard, Penny, Rajesh, Howard
	int n, c = 0;
	cin >> n;
	while (true)
	{
		if (n + 4 >= pow(2, c) * 5 && n + 4 < pow(2, c) * 6)
		{
			cout << "Sheldon";
			break;
		}
		if (n + 4 >= pow(2, c) * 6 && n + 4 < pow(2, c) * 7)
		{
			cout << "Leonard";
			break;
		}
		if (n + 4 >= pow(2, c) * 7 && n + 4 < pow(2, c) * 8)
		{
			cout << "Penny";
			break;
		}
		if (n + 4 >= pow(2, c) * 8 && n + 4 < pow(2, c) * 9)
		{
			cout << "Rajesh";
			break;
		}
		if (n + 4 >= pow(2, c) * 9 && n + 4 < pow(2, c) * 10)
		{
			cout << "Howard";
			break;
		}
		c++;
	}
	return 0;
}
