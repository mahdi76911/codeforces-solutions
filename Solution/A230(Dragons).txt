#include <iostream>

using namespace std; 

int main() 
{
	int s, n, x[1000], y[1000], m = 0;
	cin >> s >> n;
	for (int i = 0; i < n; i++)
		cin >> x[i] >> y[i];

	int c;
	for (int i = 1; i < n; i++)
	{
		c = i;
		while (c > 0)
		{
			if (x[c] < x[c - 1])
			{
				swap(x[c], x[c - 1]);
				swap(y[c], y[c - 1]);
				c--;
			}
			else
				break;
		}
	}
	for (int i = 0; i < n; i++)
	{
		if (s > x[i])
			s += y[i];
		else
		{
			cout << "NO";
			return 0;
		}
	}
	cout << "YES";
	return 0;
}
