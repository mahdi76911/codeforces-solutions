#include <iostream>
#include <string>

using namespace std;

int main()
{
	int n;
	bool b = true;
	string s, res;
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> s;
		if (b)
		{
			if (s.substr(0, 2) == "OO")
			{
				s[0] = '+';
				s[1] = '+';
				b = false;
			}
			else if (s.substr(3, 2) == "OO")
			{
				s[3] = '+';
				s[4] = '+';
				b = false;
			}
		}
		res += s + "\n";
	}
	cout << (b ? "NO" : "YES") << endl;
	cout << (b ? "" : res);
	return 0;
}
