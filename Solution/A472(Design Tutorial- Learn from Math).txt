#include <iostream>
#include <math.h>

using namespace std;

bool IsPrime(long int x)
{
	for (int i = 2; i <= sqrt(x); i++)
		if (x%i == 0)
			return false;
	return true;
}

int main()
{
	long int n;
	cin >> n;
	if (n % 2 == 0)
		cout << "4 " << n - 4;
	else
	{
		for (int i = 4; i <= (n + 1) / 2; i += 2)
		{
			if (!IsPrime(n - i))
			{
				cout << i << " " << n - i;
				break;
			}
		}
	}
	return 0;
}
