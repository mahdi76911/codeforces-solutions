#include <iostream>
#include <string>

using namespace std;

int main()
{
	int n, k, a[101];
	for (int i = 0; i < 101; i++)
		a[i] = 0;
	string x, p;
	cin >> n >> k;
	for (int i = 0; i < n; i++)
	{
		cin >> x;
		a[x.length()]++;
	}
	cin >> p;
	for (int i = 1; i < p.length(); i++)
		a[0] += a[i];
	n = a[0] + a[p.length()];
	cout << a[0] + 1 + ((a[0] / k) * 5) << " " << n + (((n - 1) / k) * 5);
	return 0;
}
