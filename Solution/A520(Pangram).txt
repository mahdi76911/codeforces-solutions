#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main()
{
	bool l[26];
	for (int i = 0; i < 26; i++)
		l[i] = true;
	int n, res = 0;
	string s;
	cin >> n >> s;
	if (n < 26)
		cout << "NO";
	else
	{
		transform(s.begin(), s.end(), s.begin(), ::tolower);
		for (int i = 0; i < n; i++)
		{
			if (l[s[i] - 'a'])
			{
				res++;
				l[s[i] - 'a'] = false;
			}
		}
		cout << ((res >= 26) ? "YES" : "NO");
	}
	return 0;
}
