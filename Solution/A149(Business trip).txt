#include <iostream>

using namespace std;

int main()
{
	int k, a[12];
	cin >> k;
	if (k == 0)
	{
		cout << 0;
		return 0;
	}
	for (int i = 0; i < 12; i++)
		cin >> a[i];
	for (int i = 1; i < 12; i++)
	{
		int c = i;
		while (c > 0 && a[c - 1] < a[c])
		{
			swap(a[c - 1], a[c]);
			c--;
		}
	}
	for (int i = 0; i < 12; i++)
	{
		k -= a[i];
		if (k <= 0)
		{
			cout << i + 1;
			return 0;
		}
	}
	cout << -1;
	return 0;
}
