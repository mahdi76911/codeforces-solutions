#include <iostream>

using namespace std;

int main()
{
	int n, k, m, x[100000];
	int s[100000];
	for (int i = 0; i < 100000; i++)
		s[i] = 0;
	cin >> n >> k >> m;
	for (int i = 0; i < n; i++)
	{
		cin >> x[i];
		s[x[i] % m]++;
	}
	for (int i = 0; i < 100000; i++)
	{
		if (s[i] >= k)
		{
			cout << "Yes" << endl;
			int jj = 0;
			while (k>0)
			{
				if (x[jj] % m == i)
				{
					cout << x[jj] << " ";
					k--;
				}
				jj++;
			}
			return 0;
		}
	}
	cout << "No";
	return 0;
}
