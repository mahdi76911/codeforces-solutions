#include <iostream>
#include <string>

using namespace std;

int main()
{
	bool B;
	string a, r = "";
	cin >> a;
	for (int i = 0; i < a.length(); i++)
	{
		B = true;
		for (int j = 0; j < r.length(); j++)
		{
			if (a[i] == r[j])
			{
				B = false;
				break;
			}
		}
		if (B)
			r += a[i];
	}
	if (r.length() % 2 == 0)
		cout << "CHAT WITH HER!";
	else
		cout << "IGNORE HIM!";
	return 0;
}
