#include <iostream>

using namespace std;

int main()
{
	int n, sum = 0, pick = 0, res = 0;
	cin >> n;
	int a[100];
	for (int i = 0; i < n; i++)
	{
		cin >> a[i];
		sum += a[i];
	}
	//sort:
	int c;
	for (int i = 1; i < n; i++)
	{
		c = i;
		while (c > 0)
		{
			if (a[c] < a[c - 1])
			{
				swap(a[c], a[c - 1]);
				c--;
			}
			else
				break;
		}
	}
	for (int i = n - 1; n >= 0, pick <= sum / 2; i--)
	{
		pick += a[i];
		res++;
	}
	cout << res;
	return 0;
}
