#include <iostream>
#include <string>

using namespace std;

int main()
{
	string a, b, x, res = "";
	int n;
	cin >> a >> b;
	res += a + " " + b + "\n";
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> x;
		if (x == a)
			cin >> a;
		else
			cin >> b;
		res += a + " " + b + "\n";
	}
	cout << res;
	return 0;
}
