#include <iostream>

using namespace std;

int main()
{
int b=0,w=0;
char x;
for(int i=0;i<64;i++)
{
cin>>x;
switch(x)
{
case 'Q':
w +=9;
break;
case 'q':
b +=9;
break;
case 'R':
w +=5;
break;
case 'r':
b +=5;
break;
case 'B':
w +=3;
break;
case 'b':
b +=3;
break;
case 'N':
w +=3;
break;
case 'n':
b +=3;
break;
case 'P':
w +=1;
break;
case 'p':
b +=1;
break;
default:
break;
}
}
if(w>b)
cout<<"White";
else if(b>w)
cout<<"Black";
else
cout<<"Draw";
return 0;
}
