#include <iostream>
#include <string>

using namespace std;

int main()
{
	int n, t;
	string s, res;
	cin >> n >> t >> s;
	res = s;
	for (int i = 0; i < t; i++)
	{
		for (int j = 0; j < n - 1; j++)
			if (s.substr(j, 2) == "BG")
			{
				res[j] = 'G';
				res[j + 1] = 'B';
			}
		s = res;
	}
	cout << s;
	return 0;
}
