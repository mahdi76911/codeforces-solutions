#include <iostream>
#include <algorithm>

using namespace std;

int MaxSumSeries(int a[], int s, int f)
{
	if (s >= f)
		return a[f];
	if (a[s + 1] == 0)
		return a[s] + MaxSumSeries(a, s + 2, f);
	int n = f - s + 1;
	if (n <= 3)
	{
		switch (n)
		{
		case 1:
			return a[s];
			break;
		case 2:
			return max(a[s], a[f]);
			break;
		case 3:
			return max(a[s + 1], a[s] + a[f]);
			break;
		default:
			break;
		}
	}
	else
		return max(a[s] + MaxSumSeries(a, s + 2, f), a[s + 1] + MaxSumSeries(a, s + 3, f));
}
