#include <iostream>

using namespace std;

int main()
{
	int n, m = 0, c = 0, x, y;
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> x >> y;
		if (x > y)
			m++;
		if (y > x)
			c++;
	}
	if (m > c)
		cout << "Mishka";
	if (c > m)
		cout << "Chris";
	if (c == m)
		cout << "Friendship is magic!^^";
	return 0;
}
