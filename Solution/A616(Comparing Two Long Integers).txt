#include <iostream>
#include <string>

using namespace std;

string ZeroDel(string num)
{
	string res = num;
	int c = 0;
	for (int i = 0; i < num.length(); i++)
	{
		if (res[i] == '0')
			c++;
		else
			break;
	}
	if (c == num.length())
		res = "0";
	else
		res = res.substr(c);
	return res;
}
int main()
{
	string a, b;
	cin >> a >> b;
	a = ZeroDel(a);
	b = ZeroDel(b);
	if (a.length() > b.length())
	{
		cout << ">";
		return 0;
	}
	if (a.length() < b.length())
	{
		cout << "<";
		return 0;
	}
	for (int i = 0; i < a.length(); i++)
	{
		if (a[i] > b[i])
		{
			cout << ">";
			return 0;
		}
		if (a[i] < b[i])
		{
			cout << "<";
			return 0;
		}
	}
	cout << "=";
	return 0;
}
