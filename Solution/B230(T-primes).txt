#include <iostream>
#include <math.h>

using namespace std;

bool isPrime(unsigned long long int x)
{
	for (unsigned long long int i = 2; i*i <= x; i++)
		if (x % i == 0)
			return false;
	return true;
}

int main()
{
	unsigned long long int x, n, s = 0;
	bool  b;
	cin >> n;
	for (int i = 0; i < n; i++)
	{
		cin >> x;
		if (s == x)
			goto End;
		s = x;
		b = true;
		if (x == 4 || x == 25 || x == 9)
			goto End;
		b = pow((int)sqrt(x), 2) == x && x != 1 && ((x % 10 == 1 && x % 9 != 0) || x % 10 == 9) && isPrime(sqrt(x));
	End:
		cout << (b? "YES" : "NO") << endl;
	}
	return 0;
}
