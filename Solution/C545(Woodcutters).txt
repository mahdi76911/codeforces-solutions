#include <iostream>

using namespace std;

int main()
{
	int n, x[100000], h[100000], s[10000], res = 2;
	cin >> n;
	cin >> x[0] >> h[0];
	for (int i = 1; i < n; i++)
	{
		cin >> x[i] >> h[i];
		s[i - 1] = (x[i] - x[i - 1]) - 1;
	}
	if (n <= 2)
	{
		cout << n;
		return 0;
	}
	for (int i = 1; i < n - 1; i++)
	{
		if (h[i] <= s[i - 1])
			res++;
		else if(h[i] <= s[i])
		{
			res++;
			s[i] -= h[i];
		}
	}
	cout << res;
	return 0;
}
